cmake_minimum_required(VERSION 3.21)
project(AlphaDecay VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG 9cca280a4d0ccf0c08f47a99aa71d1b0e52f8d03)
FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG a74efa0d5628b74adc0426af4c5710e287fa7c2c)
FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG dbb5eeaadffb6a3ba6a60de1290312e5802dba5a)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(json glfw imgui)
find_package(OpenGL REQUIRED)
add_library(imgui STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
target_include_directories(imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
target_link_libraries(imgui PUBLIC glfw OpenGL::GL)
add_library(alpha_core STATIC native/src/core.cpp native/src/potentials.cpp native/src/spherical_wkb.cpp native/src/spherical_model.cpp native/src/deformation.cpp native/src/channels.cpp)
target_include_directories(alpha_core PUBLIC native/include)
file(READ "${CMAKE_SOURCE_DIR}/observatory/isotopes.json" CATALOG_JSON)
file(READ "${CMAKE_SOURCE_DIR}/observatory/validation.json" VALIDATION_JSON)
file(READ "${CMAKE_SOURCE_DIR}/observatory/holdout-validation.json" HOLDOUT_JSON)
file(READ "${glfw_SOURCE_DIR}/LICENSE.md" GLFW_LICENSE)
file(READ "${imgui_SOURCE_DIR}/LICENSE.txt" IMGUI_LICENSE)
file(READ "${json_SOURCE_DIR}/LICENSE.MIT" JSON_LICENSE)
file(READ "${CMAKE_SOURCE_DIR}/native/khronos-license.txt" KHRONOS_LICENSE)
file(READ "${CMAKE_SOURCE_DIR}/native/proggyclean-license.txt" PROGGYCLEAN_LICENSE)
configure_file(native/src/embedded.hpp.in generated/embedded.hpp @ONLY)
add_executable(AlphaDecay native/src/main.cpp native/src/viewer.cpp native/src/channel_json.cpp native/src/gl_loader.c)
target_include_directories(AlphaDecay PRIVATE "${CMAKE_BINARY_DIR}/generated" "${glfw_SOURCE_DIR}/deps")
target_link_libraries(AlphaDecay PRIVATE alpha_core imgui nlohmann_json::nlohmann_json)
if(WIN32)
  target_link_libraries(AlphaDecay PRIVATE opengl32)
endif()
enable_testing()
add_test(NAME self-test COMMAND AlphaDecay --self-test)
add_executable(spherical_wkb_test native/tests/spherical_wkb_test.cpp)
target_link_libraries(spherical_wkb_test PRIVATE alpha_core)
add_test(NAME spherical-wkb COMMAND spherical_wkb_test)
add_executable(channels_test native/tests/channels_test.cpp)
target_link_libraries(channels_test PRIVATE alpha_core)
add_test(NAME channels COMMAND channels_test)

find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME cli-contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/native/tests/contract.py $<TARGET_FILE:AlphaDecay>)
endif()
